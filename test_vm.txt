### VALIDO - Factorial en el main (ciclico) (esperado: 120)
programa factmain;
vars n, i, res: entero;
inicio {
    n = 5;
    res = 1;
    i = 1;
    mientras (i < n) haz {
        i = i + 1;
        res = res * i;
    };
    escribe("factorial de", n, "=", res);
}
fin
 
### VALIDO - Factorial ciclico en funcion (esperado: 120)
programa factciclo;
vars r: entero;
entero factorial(n: entero) {
    vars i, res: entero;
    {
        res = 1;
        i = 1;
        mientras (i < n) haz {
            i = i + 1;
            res = res * i;
        };
        regresa res;
    }
};
inicio {
    r = factorial(5);
    escribe("factorial(5) =", r);
}
fin
 
### VALIDO - Factorial recursivo en funcion (esperado: 120)
programa factrec;
vars r: entero;
entero factorial(n: entero) {
    {
        si (n < 2) {
            regresa 1;
        } sino {
            regresa n * factorial(n - 1);
        };
    }
};
inicio {
    r = factorial(5);
    escribe("factorial(5) =", r);
}
fin
 
### VALIDO - Fibonacci en el main (ciclico) (esperado: 55)
programa fibmain;
vars n, i, a, b, t: entero;
inicio {

    b = 1;
    i = 0;
    mientras (i < n) haz {

    };
    escribe("fibonacci de", n, "=", a);
}
fin
 
### VALIDO - Fibonacci ciclico en funcion (esperado: 55)
programa fibciclo;
vars r: entero;
entero fib(n: entero) {
    vars a, b, t, i: entero;
    {
        a = 0;
        b = 1;
        i = 0;
        mientras (i < n) haz {
            t = a + b;
            a = b;
            b = t;
            i = i + 1;
        };
        regresa a;
    }
};
inicio {
    r = fib(10);
    escribe("fib(10) =", r);
}
fin
 
### VALIDO - Fibonacci recursivo en funcion (esperado: 55)
programa fibrec;
vars r: entero;
entero fib(n: entero) {
    {
        si (n < 2) {
            regresa n;
        } sino {
            regresa fib(n - 1) + fib(n - 2);
        };
    }
};
inicio {
    r = fib(10);
    escribe("fib(10) =", r);
}
fin
