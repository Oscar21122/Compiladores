### VALIDO - Programa completo (if + while)

programa test;

vars
x, y: entero;
z: flotante;

inicio
{
    x = 5;
    y = 10;
    z = 3.14;

    si (x < y) {
        escribe("x es menor", x);
    };

    mientras (x < y) haz {
        x = x + 1;
    };
}
fin

### VALIDO - If-Else

programa cond;
vars x: entero;
inicio {
    si (x == 5) {
        x = 1;
    } sino {
        x = 2;
    };
} fin
